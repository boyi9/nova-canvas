# deploy-local-ollama.ps1
Write-Host "=== 开始初始化 Docker 部署环境 ===" -ForegroundColor Yellow

# 1. 检查 Docker
if (!(Get-Command docker -ErrorAction SilentlyContinue)) {
    Write-Host "错误: Docker 未安装" -ForegroundColor Red
    exit 1
}

# 2. 检查 NVIDIA 驱动
try {
    $gpuInfo = nvidia-smi --query-gpu=name --format=csv,noheader | Select-Object -First 1
    if ($gpuInfo -like "*RTX 5070*") {
        Write-Host "检测到 RTX 5070 显卡，应用专属显存配置" -ForegroundColor Green
    } else {
        Write-Host "警告: 检测到的显卡为 $gpuInfo，非预期的 RTX 5070" -ForegroundColor Yellow
    }
} catch {
    Write-Host "错误: 未检测到 NVIDIA GPU 驱动或 nvidia-smi 不可用" -ForegroundColor Red
    exit 1
}

# 3. 创建目录 (Windows 路径)
New-Item -ItemType Directory -Force -Path "D:\ollama-models" | Out-Null
New-Item -ItemType Directory -Force -Path ".\ci-workspace" | Out-Null
New-Item -ItemType Directory -Force -Path ".\app" | Out-Null

# 4. 构建 Docker 镜像
Write-Host "正在构建 Docker 镜像..." -ForegroundColor Yellow
docker-compose build --no-cache
if ($LASTEXITCODE -ne 0) {
    Write-Host "错误: Docker 镜像构建失败" -ForegroundColor Red
    exit 1
}
Write-Host "Docker 镜像构建成功" -ForegroundColor Green

# 5. 启动容器
Write-Host "正在启动容器服务..." -ForegroundColor Yellow
docker-compose up -d
if ($LASTEXITCODE -ne 0) {
    Write-Host "错误: 容器启动失败" -ForegroundColor Red
    exit 1
}
Write-Host "容器服务启动成功" -ForegroundColor Green

# 6. 等待服务就绪
Write-Host "等待 Ollama 服务就绪..." -ForegroundColor Yellow
Start-Sleep -Seconds 10

# 7. 执行环境一致性校验
Write-Host "执行环境一致性校验..." -ForegroundColor Yellow

# 校验 GPU
$gpuCheck = docker exec local-ollama-rtx5070 nvidia-smi 2>&1
if ($gpuCheck -like "*RTX 5070*") {
    Write-Host "[PASS] GPU 识别正常" -ForegroundColor Green
} else {
    Write-Host "[FAIL] GPU 识别失败" -ForegroundColor Red
}

# 校验 Ollama 版本
$ollamaVer = docker exec local-ollama-rtx5070 ollama -v 2>&1
if ($ollamaVer -like "*0.32.13*") {
    Write-Host "[PASS] Ollama 版本匹配 (0.32.13)" -ForegroundColor Green
} else {
    Write-Host "[FAIL] Ollama 版本不匹配: $ollamaVer" -ForegroundColor Red
}

# 校验目录挂载
$dirCheck = docker exec local-ollama-rtx5070 ls /d/ollama-models 2>&1
if ($LASTEXITCODE -eq 0) {
    Write-Host "[PASS] 模型缓存目录挂载正常" -ForegroundColor Green
} else {
    Write-Host "[FAIL] 模型缓存目录挂载失败" -ForegroundColor Red
}

Write-Host "=== 所有环境校验完成 ===" -ForegroundColor Green
Write-Host "部署成功！Ollama 服务运行在 http://localhost:11434" -ForegroundColor Green
